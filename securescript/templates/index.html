QUICKSTART // RUNTIME INTEGRATION
# 1. Register a project and receive your proxy slug
curl -X POST https://securescript-pinq.onrender.com/platform/projects \
-H "Authorization: Bearer <YOUR_JWT_TOKEN>" \
-H "Content-Type: application/json" \
-d '{"name": "Production App", "frontend_url": "https://myapp.vercel.app"}'

# 2. Route your live traffic safely through SecureScript:
curl -i https://securescript-pinq.onrender.com/proxy/production-app-4c2b/
# Inline ASGI Middleware integration for FastAPI or Starlette
from securescript.middleware.asgi import SecureScriptMiddleware
from fastapi import FastAPI

app = FastAPI()
app.add_middleware(
SecureScriptMiddleware,
mode="block",
dl_threshold=0.85
)
# Deploy production multi-tenant gateway with PostgreSQL
